/**
 * C3 — Écrit avec le plus grand soin, sans accès aux décisions du projet.
 * Ni registre, ni échelle, ni seuils, ni composants : rien d'autre que
 * l'intention de faire du bon travail.
 *
 * Soin apporté volontairement : les quatre états sont traités, la hiérarchie
 * de titres est continue, l'échelle d'espacement est cohérente (base 8),
 * les messages d'erreur sont humains et offrent une porte de sortie,
 * les cibles tactiles font 44 px, le contraste est tenu.
 */
import { useRequete, useMutation } from '../design-system/donnees/useRequete.js'

export function SoigneNonInforme() {
  const factures = useRequete('/factures')
  const { lancer, enAttente, erreur } = useMutation('/relances')

  if (factures.chargement) {
    return (
      <div className="p-8">
        <div className="animate-pulse space-y-3" aria-busy="true" aria-live="polite">
          <div className="h-4 w-1/3 rounded bg-gray-200" />
          <div className="h-4 w-2/3 rounded bg-gray-200" />
        </div>
      </div>
    )
  }

  if (factures.erreur) {
    return (
      <div className="p-8">
        <div role="alert" className="rounded-lg border border-red-200 bg-red-50 p-6">
          <h2 className="text-lg font-semibold text-red-800">Les factures sont indisponibles</h2>
          <p className="mt-2 text-sm text-red-700">
            Le service de facturation ne répond pas. Aucune relance n'a été envoyée.
            Vous pouvez réessayer, ou revenir dans quelques minutes.
          </p>
        </div>
      </div>
    )
  }

  return (
    <main className="mx-auto max-w-5xl p-8">
      <header className="mb-8">
        <p className="text-xs font-semibold uppercase tracking-widest text-gray-500">Facturation</p>
        <h1 className="mt-2 text-3xl font-semibold tracking-tight text-gray-900">Suivi des factures</h1>
        <p className="mt-3 max-w-prose text-base text-gray-600">
          Les factures en cours, leur échéance et leur statut.
        </p>
      </header>

      <section aria-labelledby="filtres" className="mb-8">
        <h2 id="filtres" className="mb-3 text-lg font-semibold text-gray-900">Filtrer</h2>
        <div className="flex flex-wrap gap-4">
          <label className="flex flex-col gap-1 text-sm font-medium text-gray-700">
            Client
            <input type="text" className="min-h-[44px] rounded-md border border-gray-300 px-3" />
          </label>
          <label className="flex flex-col gap-1 text-sm font-medium text-gray-700">
            Statut
            <input type="text" className="min-h-[44px] rounded-md border border-gray-300 px-3" />
          </label>
        </div>
      </section>

      <section aria-labelledby="liste">
        <h2 id="liste" className="mb-4 text-lg font-semibold text-gray-900">Les factures</h2>
        {factures.donnees.length === 0 ? (
          <div className="rounded-lg border border-dashed border-gray-300 p-8 text-center">
            <h3 className="text-base font-semibold text-gray-900">Aucune facture en cours</h3>
            <p className="mt-2 text-sm text-gray-600">
              Rien n'est en attente de paiement. Les factures réglées n'apparaissent pas ici.
            </p>
          </div>
        ) : (
          <ul className="grid gap-4 sm:grid-cols-3">
            {factures.donnees.map((f) => (
              <li key={f.id} className="rounded-lg border border-gray-200 p-5">
                <h3 className="text-base font-semibold text-gray-900">{f.client}</h3>
                <p className="mt-1 text-sm text-gray-600">{f.montant} · échéance {f.echeance}</p>
                <p className="mt-3 text-xs font-semibold uppercase tracking-wide text-amber-700">{f.statut}</p>
              </li>
            ))}
          </ul>
        )}
      </section>

      <section aria-labelledby="agir" className="mt-8">
        <h2 id="agir" className="mb-3 text-lg font-semibold text-gray-900">Agir</h2>
        <button
          type="button"
          onClick={lancer}
          className="min-h-[44px] rounded-md bg-gray-900 px-5 text-sm font-semibold text-white"
        >
          {enAttente ? 'Envoi de la relance…' : 'Relancer le client'}
        </button>
        {erreur ? (
          <p role="status" className="mt-2 text-sm text-red-700">
            La relance n'est pas partie. Réessayer.
          </p>
        ) : null}
      </section>
    </main>
  )
}
